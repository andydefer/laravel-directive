+<?php
+
+declare(strict_types=1);
+
+namespace AndyDefer\Directive\Services;
+
+use AndyDefer\Directive\Records\ValidationResultRecord;
+
+/**
+ * Validates directive signatures for format compliance.
+ *
+ * Ensures signatures use only colon (:) or hyphen (-) as delimiters,
+ * with no special characters like @, _, or spaces.
+ *
+ * Allowed patterns:
+ * - user:list
+ * - cache-clear
+ * - api:user-profile
+ *
+ * Disallowed patterns:
+ * - user@list (invalid delimiter)
+ * - create_user (underscore not allowed)
+ * - user: (trailing delimiter)
+ * - :list (leading delimiter)
+ */
+final class SignatureValidationService
+{
+    private const DELIMITERS = [':', '-'];
+    private const ALLOWED_PATTERN = '/^[a-zA-Z][a-zA-Z0-9]*([:-][a-zA-Z][a-zA-Z0-9]*)*$/';
+
+    /**
+     * Validate a directive signature.
+     *
+     * @param string $signature The signature to validate (e.g., 'user:list')
+     * @return ValidationResultRecord Contains isValid status and error message if invalid
+     */
+    public function validate(string $signature): ValidationResultRecord
+    {
+        if (empty($signature)) {
+            return $this->invalid('Signature cannot be empty');
+        }
+
+        if (str_starts_with($signature, '--')) {
+            return $this->valid();
+        }
+
+        if (!preg_match(self::ALLOWED_PATTERN, $signature)) {
+            return $this->invalid(
+                sprintf(
+                    'Invalid signature format: "%s". Allowed: letters, numbers, colon (:), hyphen (-). No special characters like @, _, or spaces. Example: user:list or cache-clear',
+                    $signature
+                )
+            );
+        }
+
+        if ($this->hasConsecutiveDelimiters($signature)) {
+            return $this->invalid(
+                sprintf('Signature "%s" has consecutive delimiters', $signature)
+            );
+        }
+
+        return $this->valid();
+    }
+
+    /**
+     * Check if signature has consecutive delimiters.
+     *
+     * @param string $signature
+     * @return bool
+     */
+    private function hasConsecutiveDelimiters(string $signature): bool
+    {
+        return (bool) preg_match('/[' . preg_quote(implode('', self::DELIMITERS), '/') . ']{2,}/', $signature);
+    }
+
+    private function valid(): ValidationResultRecord
+    {
+        return new ValidationResultRecord(isValid: true, error: null);
+    }
+
+    private function invalid(string $error): ValidationResultRecord
+    {
+        return new ValidationResultRecord(isValid: false, error: $error);
+    }
